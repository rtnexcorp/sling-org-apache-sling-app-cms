# Apache Sling CMS - Workflow Service User Configuration

# Create service user for workflow operations
# This user is used by SimpleBPMNRepositoryService, ProcessInstanceManager, TaskManager
create service user sling-cms-workflow with path system/sling/cms

# Create workflow storage paths in JCR
create path (sling:Folder) /var/workflow
create path (sling:Folder) /var/workflow/definitions
create path (sling:Folder) /var/workflow/instances
create path (sling:Folder) /var/workflow/tasks
create path (sling:Folder) /var/workflow/history
create path (sling:Folder) /var/workflow/history/processes
create path (sling:Folder) /var/workflow/history/tasks

# Grant permissions for workflow service user
set ACL for sling-cms-workflow
    # Read access to root node for navigation
    allow jcr:read on /
    
    # Read access to /var for path creation
    allow jcr:read,rep:write on /var
    
    # Read access to content for workflow context (e.g., content paths in process variables)
    allow jcr:read on /content
    
    # Full access to workflow storage paths
    # - /var/workflow/definitions: Store process definitions (BPMN XML)
    # - /var/workflow/instances: Store process instances
    # - /var/workflow/tasks: Store user tasks
    # - /var/workflow/history: Store historical data (completed processes and tasks)
    allow jcr:all on /var/workflow
    allow jcr:all on /var/workflow/definitions
    allow jcr:all on /var/workflow/instances
    allow jcr:all on /var/workflow/tasks
    allow jcr:all on /var/workflow/history
end
